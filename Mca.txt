
MICROCONTROLLER LAB PROGRAMS (8051) 

Program 1(a): Copy a block of 10 numbers
ORG 00H
MOV R2,#0AH
MOV R0,#20H
MOV R1,#50H
NEXT: MOV A,@R0
MOV @R1,A
INC R0
INC R1
DJNZ R2,NEXT
END

XCH A,@R1
MOV @R0,A
INC R0

END

Program 1(c): Sort numbers in ascending order
ORG 00H

MOV R0,#50H
MOV R2,#04H

MOV B,A
INC R0
MOV A,@R0

SJMP SKIP
notequal: JNC SKIP
DEC R0
MOV @R0,A
INC R0
MOV @R0,B
SKIP: DJNZ R2,nxtcomp
DJNZ R3,nxtpass
END

Program 1(d): Find largest number

Program 6: DAC Waveforms
(i) Sine Wave

#include <reg52.h>

void main()
{
    unsigned char i;
    unsigned char code table[8]={128,176,218,255,218,176,128,80};

    while(1)
    {
        for(i=0;i<8;i++)
            P0=table[i];
    }
}

(ii) Square Wave

#include <reg52.h>

void MSDelay(unsigned int);

void main()
{
    P0 = 0x00;

    while(1)
    {
        P0 = 0x00;      // Logic LOW
        MSDelay(50);

        P0 = 0xFF;      // Logic HIGH
        MSDelay(50);
    }
}

void MSDelay(unsigned int value)
{
    unsigned int y;

    for(y = 0; y < value; y++);
}

(iii) Triangular Wave

#include <reg52.h>

void main()
{
    unsigned int i;

    P0 = 0x00;

    while(1)
    {
        for(i = 0; i < 255; i++)
        {
            P0 = i;
        }

        for(i = 255; i > 0; i--)
        {
            P0 = i;
        }
    }
}

(vi) Sawtooth Wave

#include <reg52.h>

void main()
{
    unsigned int i;

    P0 = 0x00;

    while(1)
    {
        for(i = 0; i < 255; i = i + 5)
        {
            P0 = i;
        }

        for(i = 255; i > 1; i--)
        {
            P0 = i;
        }
    }
}

Program 7: Stepper Motor Control


(a) Clockwise Rotation

#include <reg52.h>

void MSDelay(unsigned char value);

void main()
{
    while(1)
    {
        P0 = 0x66;
        MSDelay(1);

        P0 = 0xCC;
        MSDelay(1);

        P0 = 0x99;
        MSDelay(1);

        P0 = 0x33;
        MSDelay(1);
    }
}

void MSDelay(unsigned char value)
{
    unsigned int x, y;

    for(x = 0; x < 500; x++);
    for(y = 0; y < value; y++);
}

Program 8: DC Motor Control

(a) Clockwise Rotation

#include <reg52.h>

void MSDelay(unsigned char value);

sbit enable = P0^6;
sbit mtr_1  = P0^7;
sbit mtr_2  = P0^4;

void main()
{
    while(1)
    {
        mtr_1 = 1;
        mtr_2 = 0;
        MSDelay(50);
        mtr_1 = 0;
    }
}

void MSDelay(unsigned char value)
{
    unsigned int x,y;

    for(x=0; x<40; x++)
